    63	class AgentLoop:
    64	    def __init__(self, model: ModelClient, registry: ToolRegistry, audit: AuditLog, config: OrchestratorConfig) -> None:
    65	        self._model = model
    66	        self._registry = registry
    67	        self._audit = audit
    68	        self._config = config
    69	
    70	    @property
    71	    def model_id(self) -> str:
    72	        return self._model.model_id
    73	
    74	    def run(self, state: TurnState, context: TurnContext) -> None:
    75	        for _ in range(self._config.max_model_steps):
    76	            response = self._complete(state)
    77	            if response is None:
    78	                return
    79	            if not response.tool_calls:
    80	                state.reply = response.text
    81	                return
    82	            state.messages.append(Message(Role.ASSISTANT, response.text or "", response.tool_calls))
    83	            if not all(self._run_tool(call, state, context) for call in response.tool_calls):
    84	                return
    85	        state.end(ReasonCode.LOOP_EXHAUSTED, DecisionKind.ESCALATE)
    86	
    87	    def _record(self, state: TurnState, audit_kind: AuditKind, /, **payload: str) -> None:
    88	        self._audit.append(AuditEntry(state.trace_id, audit_kind, payload, SYSTEM_PROMPT_VERSION, self.model_id))
    89	
    90	    def _complete(self, state: TurnState) -> ModelResponse | None:
    91	        request = ModelRequest(SYSTEM_PROMPT, tuple(state.messages), self._registry.schemas())
    92	        self._record(state, AuditKind.MODEL_CALL, messages=str(len(request.messages)))
    93	        try:
    94	            return self._model.complete(request)
    95	        except Exception as error:  # noqa: BLE001 -- provider boundary: any SDK/network error ends the turn safely; logged and audited, not swallowed
    96	            logger.warning("model client failed: %s", type(error).__name__)
    97	            self._record(state, AuditKind.MODEL_ERROR, error=type(error).__name__)
    98	            state.end(ReasonCode.MODEL_UNAVAILABLE, DecisionKind.ESCALATE)
    99	            return None
   100	
   101	    def _run_tool(self, call: ToolCall, state: TurnState, context: TurnContext) -> bool:
   102	        self._record(state, AuditKind.TOOL_CALL, name=call.name, arguments=call.arguments_json)
   103	        state.tool_names.append(call.name)
   104	        outcome = self._registry.run(call, context)
   105	        if isinstance(outcome, ToolFault):
   106	            self._record(state, AuditKind.DECISION, reason=outcome.reason.value, detail=outcome.detail)
   107	            state.end(outcome.reason, DecisionKind.DENY)
   108	            return False
   109	        if outcome.proposal is not None and state.proposal is not None:
   110	            self._record(state, AuditKind.DECISION, reason=ReasonCode.SECOND_PROPOSAL.value)
   111	            state.end(ReasonCode.SECOND_PROPOSAL, DecisionKind.DENY)
   112	            return False
   113	        self._apply(call, outcome, state)
   114	        return True
   115	
   116	    def _apply(self, call: ToolCall, outcome: ToolOutcome, state: TurnState) -> None:
   117	        result = serialise(outcome.result)
   118	        self._record(state, AuditKind.TOOL_RESULT, name=call.name, result=result)
   119	        if outcome.decision is not None:
   120	            state.decisions.append(outcome.decision)
   121	            self._record(state, AuditKind.DECISION, kind=outcome.decision.kind.value, reason=outcome.decision.reason)
   122	        if outcome.escalation_reasons:
   123	            state.escalation_reasons.extend(outcome.escalation_reasons)
   124	            state.escalation_subject = _subject(call)
   125	            state.escalation_amount = outcome.decision.amount if outcome.decision else None
   126	        state.proposal = outcome.proposal or state.proposal
   127	        state.record_amounts |= amounts_in(result)
   128	        state.refund_on_record = state.refund_on_record or _shows_refund(outcome)
   129	        untrusted = json.dumps({"untrusted_data": outcome.result}, ensure_ascii=False, default=str)
   130	        state.messages.append(Message(Role.TOOL, untrusted, tool_call_id=call.call_id))

    20	def _outcome(record: RefundRecord) -> ActionOutcome:
    21	    return ActionOutcome(
    22	        refund_id=record.refund_id,
    23	        order_id=record.order_id,

    31	class RefundExecutor:
    32	    def __init__(self, engine: PolicyEngine, new_id: Callable[[], str]) -> None:
    33	        self._engine = engine
    34	        self._new_id = new_id
    35	        self._lock = threading.Lock()
    36	
    37	    def execute(
    38	        self,
    39	        proposal: ProposedRefund,
    40	        customer_id: CustomerId,
    41	        orders: CustomerOrders,
    42	        refunds: CustomerRefunds,
    43	        trace_id: TraceId,
    44	    ) -> ActionOutcome | Decision:
    45	        with self._lock:
    46	            existing = refunds.find_by_key(proposal.idempotency_key)
    47	            if existing is not None:
    48	                return _outcome(existing)
    49	            if not self._still_valid(proposal, orders, refunds):
    50	                return Decision(DecisionKind.DENY, ReasonCode.CONFIRMATION_STALE)
    51	            record = self._issue(proposal, customer_id, trace_id)
    52	            refunds.add(record)
    53	        logger.info("refund issued refund_id=%s trace_id=%s", record.refund_id, trace_id)
    54	        return _outcome(record)

    56	    def _still_valid(self, proposal: ProposedRefund, orders: CustomerOrders, refunds: CustomerRefunds) -> bool:
    57	        check = RefundCheck(
    58	            order=orders.find_order(proposal.order_id),
    59	            item_id=proposal.item_id,
    60	            quantity=proposal.quantity,
    61	            claimed_amounts=(),
    62	            order_refunds=refunds.for_order(proposal.order_id),
    63	            customer_refunds=refunds.all(),
    64	        )
    65	        decision = self._engine.assess_refund(check).decision
    66	        return decision.kind is DecisionKind.NEEDS_CONFIRMATION and decision.amount == proposal.amount

    70	            refund_id=RefundId(f"{REFUND_ID_PREFIX}{self._new_id()}"),
    71	            idempotency_key=proposal.idempotency_key,
    72	            customer_id=customer_id,
    73	            order_id=proposal.order_id,
    74	            item_id=proposal.item_id,
    75	            quantity=proposal.quantity,
    76	            amount=proposal.amount,

    38	    def __init__(

    52	    def handle(self, request: SupportRequest) -> TurnResult:
    53	        trace_id = TraceId(f"trace-{self._new_id()}")
    54	        self._record(trace_id, AuditKind.REQUEST, customer_id=request.customer_id, message=request.message)
    55	        key: ConversationKey = (request.conversation_id, request.customer_id)
    56	        pending = self._pending.take(key)
    57	        if pending is not None and self._is_affirmation(request.message):
    58	            result = self._confirm(request, pending, trace_id)
    59	        else:
    60	            result = self._converse(request, key, trace_id)
    61	        self._conversations.append(key, Message(Role.USER, request.message), Message(Role.ASSISTANT, result.reply))
    62	        decisions = ", ".join(f"{decision.kind}:{decision.reason}" for decision in result.decisions)
    63	        self._record(trace_id, AuditKind.RESULT, reply=result.reply, decisions=decisions)
    64	        return result

    66	    def _record(self, trace_id: TraceId, audit_kind: AuditKind, /, **payload: str) -> None:
    67	        entry = AuditEntry(trace_id, audit_kind, payload, SYSTEM_PROMPT_VERSION, self._loop.model_id)
    68	        self._audit.append(entry)

    70	    def _is_affirmation(self, message: str) -> bool:
    71	        return message.strip().lower().rstrip(".!") in self._config.affirmations
    72	
    73	    def _confirm(self, request: SupportRequest, proposal: ProposedRefund, trace_id: TraceId) -> TurnResult:
    74	        context = self._context.build(request.customer_id)
    75	        outcome = self._executor.execute(proposal, request.customer_id, context.orders, context.refunds, trace_id)
    76	        if isinstance(outcome, ActionOutcome):
    77	            self._record(trace_id, AuditKind.ACTION, refund_id=outcome.refund_id, amount=outcome.amount.format())
    78	            decision = Decision(DecisionKind.ALLOW, ReasonCode.CONFIRMED, outcome.amount)
    79	            reply = replies.executed_line(outcome, proposal.item_name)
    80	            return TurnResult(reply, trace_id, (), (decision,), None, (outcome,), None)
    81	        self._record(trace_id, AuditKind.DECISION, kind=outcome.kind, reason=outcome.reason)
    82	        return TurnResult(replies.STALE_CONFIRMATION, trace_id, (), (outcome,), None, (), None)
    83	
    84	    def _converse(self, request: SupportRequest, key: ConversationKey, trace_id: TraceId) -> TurnResult:
    85	        claims = extract_claimed_amounts(request.message, self._config.default_currency)
    86	        context = self._context.build(request.customer_id, claims)
    87	        history = [*self._conversations.history(key), Message(Role.USER, request.message)]
    88	        state = TurnState(trace_id, history)
    89	        self._loop.run(state, context)
    90	        reply = self._compose(state, request, context)
    91	        escalation = self._escalation(state, request)
    92	        if state.proposal is not None:
    93	            self._pending.put(key, state.proposal)
    94	        decisions = tuple(state.decisions) or (NO_ACTION,)
    95	        return TurnResult(reply, trace_id, tuple(state.tool_names), decisions, state.proposal, (), escalation)
    96	
    97	    def _model_text(self, state: TurnState, request: SupportRequest, context: TurnContext) -> str:
    98	        if state.ended_by is not None:
    99	            return "" if state.escalation_reasons else replies.CLARIFY
   100	        own_ids = {order.order_id for order in context.orders.list_orders()}
   101	        allowed = AllowedFacts(
   102	            order_ids=frozenset(own_ids | set(self._order_ids.findall(request.message))),
   103	            amounts=frozenset(state.record_amounts),
   104	            refund_on_record=state.refund_on_record,
   105	        )
   106	        text = (state.reply or "").strip()
   107	        blocked = self._check.check(text, allowed) if text else ReasonCode.OUTPUT_BLOCKED
   108	        if blocked is None:
   109	            return text
   110	        self._record(state.trace_id, AuditKind.OUTPUT_CHECK, reason=blocked, reply=text)
   111	        state.decisions.append(Decision(DecisionKind.DENY, blocked))
   112	        state.proposal = None
   113	        return replies.SAFE_FALLBACK
   114	
   115	    def _compose(self, state: TurnState, request: SupportRequest, context: TurnContext) -> str:
   116	        lines = [self._model_text(state, request, context)]
   117	        if state.proposal is not None:
   118	            lines.append(replies.confirmation_line(state.proposal))
   119	        dispute = self._dispute(state, context.claimed_amounts)
   120	        if dispute:
   121	            lines.append(dispute)
   122	        if state.escalation_reasons:
   123	            lines.append(replies.escalation_line(state.trace_id))
   124	        return " ".join(line for line in lines if line)
   125	
   126	    def _dispute(self, state: TurnState, claims: tuple[Money, ...]) -> str:
   127	        if ReasonCode.AMOUNT_DISPUTE not in state.escalation_reasons or not claims:
   128	            return ""
   129	        record_amount = state.escalation_amount
   130	        return replies.dispute_line(max(claims, key=lambda m: m.amount), record_amount) if record_amount else ""
   131	
   132	    def _escalation(self, state: TurnState, request: SupportRequest) -> Escalation | None:
   133	        if not state.escalation_reasons:
   134	            return None
   135	        reasons = tuple(dict.fromkeys(state.escalation_reasons))
   136	        amount = state.escalation_amount.format() if state.escalation_amount else "none"
   137	        summary = f"reasons={','.join(reasons)}; {state.escalation_subject or 'no order'}; amount={amount}"
   138	        self._record(state.trace_id, AuditKind.ESCALATION, summary=summary)
   139	        return Escalation(reasons, request.customer_id, summary, state.trace_id)
