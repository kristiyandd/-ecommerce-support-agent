    54	class AgentLoop:
    55	    def __init__(
    56	        self, model: ModelClient, registry: ToolRegistry, auditor: Auditor, config: OrchestratorConfig
    57	    ) -> None:
    58	        self._model = model
    59	        self._registry = registry
    60	        self._auditor = auditor
    61	        self._config = config
    62	
    63	    def run(self, state: TurnState, context: TurnContext) -> None:
    64	        for _ in range(self._config.max_model_steps):
    65	            response = self._complete(state)
    66	            if response is None:
    67	                return
    68	            if not response.tool_calls:
    69	                state.reply = response.text
    70	                return
    71	            state.messages.append(Message(Role.ASSISTANT, response.text or "", response.tool_calls))
    72	            if not all(self._run_tool(call, state, context) for call in response.tool_calls):
    73	                return
    74	        state.end(ReasonCode.LOOP_EXHAUSTED, DecisionKind.ESCALATE)
    75	
    76	    def _complete(self, state: TurnState) -> ModelResponse | None:
    77	        request = ModelRequest(SYSTEM_PROMPT, tuple(state.messages), self._registry.schemas())
    78	        self._auditor.record(state.trace_id, AuditKind.MODEL_CALL, messages=len(request.messages))
    79	        try:
    80	            return self._model.complete(request)
    81	        except Exception as error:  # provider boundary: any SDK or network error ends the turn safely
    82	            logger.exception("model client failed")
    83	            self._auditor.record(state.trace_id, AuditKind.MODEL_ERROR, error=type(error).__name__)
    84	            state.end(ReasonCode.MODEL_UNAVAILABLE, DecisionKind.ESCALATE)
    85	            return None

    87	    def _run_tool(self, call: ToolCall, state: TurnState, context: TurnContext) -> bool:
    88	        self._auditor.record(state.trace_id, AuditKind.TOOL_CALL, name=call.name, arguments=call.arguments_json)
    89	        state.tool_names.append(call.name)
    90	        outcome = self._registry.run(call, context)
    91	        if isinstance(outcome, ToolFault):
    92	            self._auditor.record(state.trace_id, AuditKind.DECISION, reason=outcome.reason, detail=outcome.detail)
    93	            state.end(outcome.reason, DecisionKind.DENY)
    94	            return False
    95	        if outcome.proposal is not None and state.proposal is not None:
    96	            self._auditor.record(state.trace_id, AuditKind.DECISION, reason=ReasonCode.SECOND_PROPOSAL)
    97	            state.end(ReasonCode.SECOND_PROPOSAL, DecisionKind.DENY)
    98	            return False
    99	        self._apply(call, outcome, state)
   100	        return True

   103	        self._auditor.record(state.trace_id, AuditKind.TOOL_RESULT, name=call.name, result=serialise(outcome.result))
   104	        if outcome.decision is not None:
   105	            state.decisions.append(outcome.decision)
   106	            self._auditor.record(
   107	                state.trace_id, AuditKind.DECISION, kind=outcome.decision.kind, reason=outcome.decision.reason
   108	            )
   109	        state.escalations.extend(outcome.escalations)
   110	        state.proposal = outcome.proposal or state.proposal
   111	        state.refund_on_record = state.refund_on_record or outcome.shows_refund

   113	        state.messages.append(Message(Role.TOOL, untrusted, tool_call_id=call.call_id))

    36	        with self._lock:
    37	            existing = context.refunds.find_by_key(proposal.idempotency_key)

    44	class SupportService:
    45	    def __init__(
    46	        self,

    59	    def handle(self, request: SupportRequest) -> TurnResult:
    60	        trace_id = TraceId(f"{TRACE_ID_PREFIX}{self._new_id()}")
    61	        self._auditor.record(trace_id, AuditKind.REQUEST, customer_id=request.customer_id, message=request.message)
    62	        try:
    63	            result = self._dispatch(request, trace_id)
    64	        except Exception as error:
    65	            logger.exception("turn failed trace_id=%s", trace_id)
    66	            self._auditor.record(trace_id, AuditKind.INTERNAL_ERROR, error=type(error).__name__)
    67	            result = self._failed(request, trace_id)
    68	        key: ConversationKey = (request.conversation_id, request.customer_id)
    69	        self._state.conversations.append(
    70	            key, Message(Role.USER, request.message), Message(Role.ASSISTANT, result.reply)
    71	        )
    72	        labels = ", ".join(decision.label() for decision in result.decisions)
    73	        self._auditor.record(trace_id, AuditKind.RESULT, reply=result.reply, decisions=labels)
    74	        return result
    75	
    76	    def _dispatch(self, request: SupportRequest, trace_id: TraceId) -> TurnResult:
    77	        key: ConversationKey = (request.conversation_id, request.customer_id)
    78	        pending = self._state.pending.take(key)
    79	        if pending is not None and self._is_affirmation(request.message):
    80	            return self._confirm(request, pending, trace_id)
    81	        return self._converse(request, key, trace_id)
    82	
    83	    def _is_affirmation(self, message: str) -> bool:
    84	        return message.strip().lower().rstrip(".!") in self._config.affirmations
    85	
    86	    def _failed(self, request: SupportRequest, trace_id: TraceId) -> TurnResult:
    87	        item = EscalationItem(ReasonCode.INTERNAL_ERROR)
    88	        escalation = Escalation((item.reason,), request.customer_id, replies.escalation_summary((item,)), trace_id)
    89	        reply = f"{replies.reason_line(ReasonCode.INTERNAL_ERROR)} {replies.escalation_line(trace_id)}"
    90	        decision = Decision(DecisionKind.ESCALATE, ReasonCode.INTERNAL_ERROR)
    91	        return TurnResult(reply, trace_id, (), (decision,), None, (), escalation)
    92	
    93	    def _confirm(self, request: SupportRequest, proposal: ProposedRefund, trace_id: TraceId) -> TurnResult:
    94	        context = self._pipeline.context.build(request.customer_id, ())
    95	        outcome = self._pipeline.executor.execute(proposal, context, trace_id)
    96	        if isinstance(outcome, ActionOutcome):
    97	            decision = Decision(DecisionKind.ALLOW, ReasonCode.CONFIRMED, outcome.amount)
    98	            reply = replies.executed_line(outcome, proposal.item_name)
    99	            result = TurnResult(reply, trace_id, (), (decision,), None, (outcome,), None)
   100	            self._auditor.record(trace_id, AuditKind.ACTION, refund_id=outcome.refund_id, replayed=outcome.replayed)
   101	            return result
   102	        self._auditor.record(trace_id, AuditKind.DECISION, kind=outcome.kind, reason=outcome.reason)
   103	        return TurnResult(replies.reason_line(outcome.reason), trace_id, (), (outcome,), None, (), None)
   104	
   105	    def _converse(self, request: SupportRequest, key: ConversationKey, trace_id: TraceId) -> TurnResult:
   106	        claims = extract_claimed_amounts(request.message, self._pipeline.context.currency)
   107	        context = self._pipeline.context.build(request.customer_id, claims)
   108	        state = TurnState(trace_id, [*self._state.conversations.history(key), Message(Role.USER, request.message)])
   109	        self._pipeline.loop.run(state, context)
   110	        model_text = self._checked_model_text(state, request, context)
   111	        reply = self._compose(model_text, state, context)
   112	        escalation = self._escalation(state, request)
   113	        if state.proposal is not None:
   114	            self._state.pending.put(key, state.proposal)
   115	        decisions = tuple(state.decisions) or (NO_ACTION,)
   116	        return TurnResult(reply, trace_id, tuple(state.tool_names), decisions, state.proposal, (), escalation)
   117	
   118	    def _checked_model_text(self, state: TurnState, request: SupportRequest, context: TurnContext) -> str:
   119	        if state.ended_by is not None:
   120	            return "" if state.escalations else replies.CLARIFY
   121	        text = (state.reply or "").strip()
   122	        has_code_lines = state.proposal is not None or bool(state.escalations) or bool(state.reasons_to_explain())
   123	        if not text:
   124	            return "" if has_code_lines else self._block(state, text)
   125	        check = self._pipeline.output_check
   126	        own_ids = frozenset(order.order_id for order in context.orders.list_orders())
   127	        allowed = AllowedFacts(own_ids | check.canonical_order_ids(request.message), state.refund_on_record)
   128	        return text if check.check(text, allowed) is None else self._block(state, text)
   129	
   130	    def _block(self, state: TurnState, text: str) -> str:
   131	        self._auditor.record(state.trace_id, AuditKind.OUTPUT_CHECK, reason=ReasonCode.OUTPUT_BLOCKED, reply=text)
   132	        state.decisions.append(Decision(DecisionKind.DENY, ReasonCode.OUTPUT_BLOCKED))
   133	        state.proposal = None
   134	        return replies.SAFE_FALLBACK
   135	
   136	    def _compose(self, model_text: str, state: TurnState, context: TurnContext) -> str:
   137	        lines = [model_text]
   138	        if model_text != replies.SAFE_FALLBACK:
   139	            lines.extend(replies.reason_line(reason) for reason in state.reasons_to_explain())
   140	        if state.proposal is not None:
   141	            lines.append(replies.confirmation_line(state.proposal))
   142	        lines.extend(self._dispute_lines(state, context))
   143	        if state.escalations:
   144	            lines.append(replies.escalation_line(state.trace_id))
   145	        return " ".join(dict.fromkeys(line for line in lines if line))
   146	
   147	    def _dispute_lines(self, state: TurnState, context: TurnContext) -> list[str]:
   148	        if not context.claimed_amounts:
   149	            return []
   150	        claimed = max(context.claimed_amounts, key=lambda money: money.amount)
   151	        disputes = (item for item in state.escalations if item.reason is ReasonCode.AMOUNT_DISPUTE)
   152	        return [replies.dispute_line(claimed, item.amount) for item in disputes if item.amount is not None]
   153	
   154	    def _escalation(self, state: TurnState, request: SupportRequest) -> Escalation | None:
   155	        if not state.escalations:
   156	            return None
   157	        items = tuple(dict.fromkeys(state.escalations))
   158	        summary = replies.escalation_summary(items)
   159	        self._auditor.record(state.trace_id, AuditKind.ESCALATION, summary=summary)
   160	        reasons = tuple(dict.fromkeys(item.reason for item in items))
   161	        return Escalation(reasons, request.customer_id, summary, state.trace_id)
